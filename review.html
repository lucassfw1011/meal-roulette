async function submitReview() { const restaurant = document.getElementById("restaurantInput").value.trim(); const dish = document.getElementById("dishInput").value.trim(); const review = document.getElementById("reviewInput").value.trim(); const status = document.getElementById("status"); if (!restaurant) { status.style.color = "#f87171"; status.innerText = "Please select or type a restaurant name!"; return; } if (selectedRating === 0 && !dish && !review) { status.style.color = "#f87171"; status.innerText = "Please provide a star rating, dish, or comment!"; return; } status.style.color = "#38bdf8"; status.innerText = "Submitting review..."; // Format payload as URL Search Params to bypass CORS preflight issues smoothly const payload = new URLSearchParams({ restaurant: restaurant, rating: selectedRating, dishes: dish, review: review }); try { await fetch(APPS_SCRIPT_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body: payload.toString() }); status.style.color = "#4ade80"; status.innerText = "✅ Thank you! Review submitted successfully."; // Append to local cache and refresh list UI allReviewsCache.push({ Restaurant: restaurant, Rating: selectedRating, RecommendedDishes: dish, ReviewText: review }); onRestaurantChanged(); // Reset Form Fields selectedRating = 0; document.querySelectorAll('.star-rating .star').forEach(s => s.classList.remove('active')); document.getElementById("dishInput").value = ""; document.getElementById("reviewInput").value = ""; } catch (err) { status.style.color = "#f87171"; status.innerText = "Error submitting review. Please try again."; } }
