async function loadPixelAssets() { if (PIXEL_ASSETS_TAB_GID === 'YOUR_PIXEL_ASSETS_TAB_GID_HERE') return; try { const res = await fetch(PIXEL_ASSETS_CSV_URL); const text = await res.text(); const lines = text.split(/\r?\n/).filter(l => l.trim()); if (lines.length <= 1) return; const leftBox = document.getElementById('pixelSidebarLeft'); const rightBox = document.getElementById('pixelSidebarRight'); const topBox = document.getElementById('pixelTop'); const bottomBox = document.getElementById('pixelBottom'); const bgBox = document.getElementById('pixelBg'); if (leftBox) leftBox.innerHTML = ''; if (rightBox) rightBox.innerHTML = ''; if (topBox) topBox.innerHTML = ''; if (bottomBox) bottomBox.innerHTML = ''; if (bgBox) { bgBox.style.backgroundImage = 'none'; } lines.slice(1).forEach(line => { const c = parseCSVLine(line); const url = c[0]; const side = (c[1] || 'Left').toLowerCase(); const animation = (c[2] || 'Float').toLowerCase(); const size = c[3] || '80'; const speed = c[4] || '6'; if (!url || !url.startsWith('http')) return; // Handle Full Background if (side === 'background' || side === 'bg') { if (bgBox) { bgBox.style.backgroundImage = `url('${url}')`; } return; } // Handle Side/Top/Bottom Floating Assets const img = document.createElement('img'); img.src = url; img.className = `pixel-asset anim-${animation}`; img.style.width = `${size}px`; img.style.animationDuration = `${speed}s`; if (side === 'right') { if (rightBox) rightBox.appendChild(img); } else if (side === 'top') { if (topBox) topBox.appendChild(img); } else if (side === 'bottom') { if (bottomBox) bottomBox.appendChild(img); } else { if (leftBox) leftBox.appendChild(img); } }); } catch (e) { console.error("Error loading pixel assets from sheet:", e); } }
